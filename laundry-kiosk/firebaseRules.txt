rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // =================================================================
    // 1. ROLE HELPERS
    // =================================================================
    
    // Checks if the user is YOUR specific Admin Account
    function isAdmin() {
       return request.auth != null && request.auth.uid == "KfyXVZ9IwMcsuE8oPVeiXqFuSDB3";
    }

    // Checks if the user is the System (Kiosk) logged in Anonymously
    function isSystem() {
       return request.auth != null && request.auth.token.firebase.sign_in_provider == 'anonymous';
    }

    // =================================================================
    // 2. COLLECTION RULES
    // =================================================================

    // --- TRANSACTIONS ---
    // System: Creates new transactions.
    // Admin: Sees all, Updates status (e.g. to 'completed').
    match /transactions/{document} {
      allow read: if isAdmin() || isSystem();
      allow create: if isSystem(); 
      allow update: if isAdmin() || isSystem();
      allow delete: if false; 
    }

    // --- LOCKERS ---
    // System: Updates status (Occupied/Free) automatically.
    // Admin: Can Reset lockers if stuck.
    match /lockers/{document} {
      allow read: if isAdmin() || isSystem();
      allow create: if isSystem();
      allow update: if isAdmin() || isSystem();
    }

    // --- OVERDUE LOGS ---
    match /overdue_logs/{document} {
      allow read: if isAdmin() || isSystem();
      allow create: if isSystem();
      allow update: if isAdmin() || isSystem();
    }

    // --- SETTINGS ---
    // Admin: Controls prices & shop name.
    // System: Reads them to display to customers.
    match /settings/{document} {
      allow read: if isAdmin() || isSystem();
      allow update: if isAdmin(); 
      allow create: if isSystem(); 
    }

    // --- USERS (Flutter App Profiles) ---
    // Admin: Sees all users.
    // Standard Users: See only their own profile.
    match /users/{userId} {
      allow read, update: if isAdmin() || (request.auth != null && request.auth.uid == userId);
      allow create: if request.auth != null; 
    }
  }
}
